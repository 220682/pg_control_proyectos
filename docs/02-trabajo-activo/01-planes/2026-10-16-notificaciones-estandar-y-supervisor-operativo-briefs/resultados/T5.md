## Contenido final de la migración 109 (para revisar)

```sql
-- Estado Devuelto del requerimiento (plan Notificaciones, T5).
-- Aditiva e idempotente. Aplicar despues de 022_rq_estados_adjuntos.sql (restriccion de estado)
-- y 108_notificaciones_evento_entorno.sql. NO se aplica desde el codigo: la revisa y aplica Victor.
--
-- 1) El estado DEVUELTO pasa a ser valido: quien puede aprobar devuelve un RQ Solicitado con un motivo;
--    el solicitante lo corrige y lo reenvia (vuelve a Solicitado). Un Devuelto no avanza ni se aprueba.
-- 2) Se guardan el motivo (texto, hasta 500 caracteres), quien lo devolvio y cuando.
--    Los tres quedan en la fila del RQ; al reenviar, el motivo tambien se copia como comentario del RQ
--    (requerimiento_comentarios), asi no se pierde cuando una nueva devolucion lo reemplace.
-- 3) Los RQ existentes no cambian: ninguno esta Devuelto y las tres columnas nuevas quedan en nulo.

alter table proyecto_requerimientos
  add column if not exists devuelto_motivo text,
  add column if not exists devuelto_por_id uuid,
  add column if not exists devuelto_en timestamptz;

-- Referencia a perfiles con el nombre que el codigo usa para traer el nombre de quien devolvio
-- (proyecto_requerimientos_devuelto_por_id_fkey). Idempotente: solo se crea si falta.
do $$
begin
  if not exists (
    select 1 from pg_constraint where conname = 'proyecto_requerimientos_devuelto_por_id_fkey'
  ) then
    alter table proyecto_requerimientos
      add constraint proyecto_requerimientos_devuelto_por_id_fkey
      foreign key (devuelto_por_id) references perfiles (id) on delete set null;
  end if;
end $$;

-- El motivo, si existe, no pasa de 500 caracteres (la API ya lo valida; esto protege la base).
do $$
begin
  if not exists (
    select 1 from pg_constraint where conname = 'proyecto_requerimientos_devuelto_motivo_largo'
  ) then
    alter table proyecto_requerimientos
      add constraint proyecto_requerimientos_devuelto_motivo_largo
      check (devuelto_motivo is null or char_length(devuelto_motivo) <= 500);
  end if;
end $$;

-- Restriccion de estado: la de 022 mas DEVUELTO.
alter table proyecto_requerimientos
  drop constraint if exists proyecto_requerimientos_estado_check;

alter table proyecto_requerimientos
  add constraint proyecto_requerimientos_estado_check
  check (estado in ('SOLICITADO', 'APROBADO', 'RECEPCIONADO', 'EN_ALMACEN', 'ATENDIDO', 'DEVUELTO'));

-- Comprobaciones (ejecutar a mano despues de aplicar; no forman parte de la migracion):
-- a) select column_name, data_type from information_schema.columns
--     where table_name = 'proyecto_requerimientos'
--       and column_name in ('devuelto_motivo', 'devuelto_por_id', 'devuelto_en');   -- 3 filas
-- b) select pg_get_constraintdef(oid) from pg_constraint
--     where conname = 'proyecto_requerimientos_estado_check';                      -- incluye 'DEVUELTO'
-- c) select estado, count(*) from proyecto_requerimientos group by estado order by estado;  -- sin Devueltos
-- d) select conname from pg_constraint
--     where conname = 'proyecto_requerimientos_devuelto_por_id_fkey';               -- 1 fila
```
